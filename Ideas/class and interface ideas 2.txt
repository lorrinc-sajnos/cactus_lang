//Interface
interface Yunk<T*> {
    i32 yunk(T* this, i32 input, i32 long_input, i32 anouther_input);
}


//Class, that implements an interface
class Foo : Yunk {
    @private bool _yippe;


    //Constructor, returns T
    Foo(){
        this._yippie = false;
    }

    bool yippie(this){
        return _yippie;
    }

    //Implementation of interface
    i32 yunk(Foo* this, i32 input, i32 long_input, i32 anouther_input)) {
        if (this._yippie)
            return input * -1 * long_input + anouther_input;
        else
            return 0;
    }
}

//Class is treated as an interface
class Bar: Foo {
    @private Foo* _foo;

    Bar(){
        _foo = new Foo();
    }


    bool yippie() from _foo;        //from keyword: calls the matching function
    //Equivalent to
    //ch08 func(i32 a, bool b) => _foo.func(a,b);


    i32 yunk(...) from _foo;
    //if using 'from'; can use ... instead of writing out the parameters,
}



class Bar_B: Foo {
    @private Foo* _foo;

    new Bar(){
        _foo = new Foo();
    }

}
